import java.util.*;

/**
 * <h1>소수 찾기</h1>
 * <h5>분류: 완전탐색</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>주어진 숫자 배열의 일부를 선택하여 조합하고 순서를 조정하여 숫자들을 만들었을 때,
 * 해당 숫자가 소수인지 판별하고 그 소수인 숫자들을 세는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>숫자 배열을 가장 내림차순으로 정렬하여 가장 큰 수를 찾고, 그 수의 소수들을 찾는다.</li>
 *     <li>숫자 배열에서 나올 수 있는 조합을 구한다.</li>
 *     <li>구한 조합을 가지고 순서를 조정하여 숫자들을 만든다.</li>
 *     <li>순서를 조정한 숫자들이 소수인지 판별하고, 소수라면 카운팅한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution("17"));
        System.out.println(solution("011"));
    }

    static int solution(String numbers) {
        int answer = 0;
        Integer[] numberArray = new Integer[numbers.length()];
        int maxNumber = 0;
        Set<Integer> primeSet = new HashSet<>();
        List<Integer[]> combinationList = new ArrayList<>();
        Set<Integer> permutationSet = new HashSet<>();

        for (int i = 0; i < numberArray.length; i++) {
            numberArray[i] = Integer.parseInt("" + numbers.charAt(i));
        }

        Arrays.sort(numberArray, Comparator.reverseOrder());

        for (int i = 0; i < numberArray.length; i++) {
            maxNumber += numberArray[i] * (int) Math.pow(10, numberArray.length - i - 1);
        }

        collectPrimes(maxNumber, primeSet);
        collectCombinations(numberArray, combinationList);
        collectPermutations(combinationList, permutationSet);

        for (int num: permutationSet) {
            if (isPrime(num, primeSet)) {
                answer++;
            }
        }

        return answer;
    }

    static void collectPrimes(int number, Set<Integer> primeSet) {
        primeSet.add(2);

        for (int i = 0; i <= Math.sqrt(number); i++) {
            if (isPrime(i, primeSet)) {
                primeSet.add(i);
            }
        }
    }

    static boolean isPrime(int number, Set<Integer> primeSet) {
        if (number <= 1) {
            return false;
        } else if (number == 2) {
            return true;
        }

        for (int prime : primeSet) {
            if (number == prime) {
                return true;
            } else if (number % prime == 0) {
                return false;
            }
        }

        return true;
    }

    static void collectCombinations(Integer[] array, List<Integer[]> combinationList) {
        for (int i = 1; i <= array.length; i++) {
            Integer[] currentArray = new Integer[i];

            combine(array, array.length, i, currentArray, 0, 0, combinationList);
        }
    }

    static void combine(
            Integer[] array,
            int n,
            int k,
            Integer[] currentArray,
            int start,
            int current,
            List<Integer[]> combinationList
    ) {
        if (current == k) {
            combinationList.add(currentArray);

            return;
        }

        for (int i = start; i < n; i++) {
            currentArray[current] = array[i];

            combine(array, n, k, Arrays.copyOf(currentArray, k), i + 1, current + 1, combinationList);
        }
    }

    static void collectPermutations(List<Integer[]> list, Set<Integer> permutationSet) {
        for (Integer[] e: list) {
            permute(e, 0, e.length - 1, permutationSet);
        }
    }

    static void permute(Integer[] array, int l, int r, Set<Integer> permutationSet) {
        if (l == r) {
            int num = 0;

            for (int i = 0; i < array.length; i++) {
                num += array[i] * (int) Math.pow(10, array.length - i - 1);
            }

            permutationSet.add(num);

            return;
        }

        for (int i = l; i <= r; i++) {
            int temp = array[i];
            array[i] = array[l];
            array[l] = temp;

            permute(array, l + 1, r, permutationSet);

            temp = array[i];
            array[i] = array[l];
            array[l] = temp;
        }
    }
}
