/**
 * <h1>카펫</h1>
 * <h5>분류: 완전탐색</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>색깔 별로 타일들을 조합했을 때 카펫의 가로, 세로 길이를 구하는 문제이다.</li>
 * <li>가로의 길이를 1씩 빼고 세로의 길이를 1씩 더하고,
 * 그 곱이 안쪽의 타일의 수와 같은지 비교하여 값을 구했다.</li>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        int[] result = solution(10, 2);

        System.out.println(result[0] + ", " + result[1]);

        result = solution(8, 1);

        System.out.println(result[0] + ", " + result[1]);

        result = solution(24, 24);

        System.out.println(result[0] + ", " + result[1]);
    }

    public static int[] solution(int brown, int yellow) {
        int[] answer = {};
        int widthHeight = (brown - 4) / 2;
        int height = 1;
        int width = widthHeight - height;

        while (width * height != yellow) {
            width--;
            height++;
        }

        int max = Math.max(width, height) + 2;
        int min = Math.min(width, height) + 2;
        answer = new int[]{max, min};

        return answer;
    }
}
