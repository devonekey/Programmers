/**
 * <h1>폰켓몬</h1>
 * <h5>분류: 해시</h5>
 * <h5>난이도: Level 1</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>N개의 폰켓몬 번호 중에 최대 N/2개까지 번호를 선택했을 때,
 * 나올 수 있는 가짓수 중 최댓값을 묻는 문제이다.</li>
 * <li>폰켓몬 번호 중복을 무시하고 존재 여부 자체를 카운팅하여 해결했다.</li>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println("answer: " + solution(new int[]{3, 1, 2, 3}));
        System.out.println("answer: " + solution(new int[]{3, 3, 3, 2, 2, 4}));
        System.out.println("answer: " + solution(new int[]{3, 3, 3, 2, 2, 2}));
    }

    public static int solution(int[] nums) {
        int answer = 0;
        boolean[] cache = new boolean[200001];

        for (int num : nums) {
            cache[num] = true;
        }

        for (boolean c : cache) {
            if (c && answer < (nums.length / 2)) {
                answer++;
            }
        }

        return answer;
    }
}
