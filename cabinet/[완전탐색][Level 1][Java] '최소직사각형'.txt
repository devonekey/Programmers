/**
 * <h1>최소직사각형</h1>
 * <h5>분류: 완전탐색</h5>
 * <h5>난이도: Level 1</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>명함의 가로, 세로가 주어질 때 이를 모두 포함시킬 수 있는 지갑의 최소 크기를 구하는 문제이다.</li></li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>가로와 세로 중에 가장 긴 길이를 구하고 가로인지 세로인지 기억한다.</li>
 *     <li>나머지 짧은 길이들 중에서 가장 큰 길이를 구한다.</li>
 * </ol>
 * <li>for문 한 번으로 처리할 수 있는데, 비효율적으로 풀었다.</li>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution(new int[][]{{60, 50}, {30, 70}, {60, 30}, {80, 40}}));
        System.out.println(solution(new int[][]{{10, 7}, {12, 3}, {8, 15}, {14, 7}, {5, 15}}));
        System.out.println(solution(new int[][]{{14, 4}, {19, 6}, {6, 16}, {18, 7}, {7, 11}}));
    }

    public static int solution(int[][] sizes) {
        int answer = 0;
        int maxLength = 0;
        boolean isWidth = false;

        for (int[] s: sizes) {
            if (s[0] > maxLength) {
                maxLength = s[0];
                isWidth = true;
            }

            if (s[1] > maxLength) {
                maxLength = s[1];
                isWidth = false;
            }
        }

        int otherLength = 0;

        for (int[] s: sizes) {
            if (isWidth) {
                if (s[0] > s[1]) {
                    otherLength = Math.max(otherLength, s[1]);
                } else {
                    otherLength = Math.max(otherLength, s[0]);
                }
            } else {
                if (s[0] < s[1]) {
                    otherLength = Math.max(otherLength, s[0]);
                } else {
                    otherLength = Math.max(otherLength, s[1]);
                }
            }
        }

        answer = maxLength * otherLength;

        return answer;
    }
}
