import java.util.*;

/**
 * <h1>섬 연결하기</h1>
 * <h5>분류: 탐욕</h5>
 * <h5>난이도: Level 3</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>주어진 모든 섬들을 최소 비용으로 연결하는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>모든 섬 연결 정보들을 비용 기준으로 오름차순 정렬한다.</li>
 *     <li>연결 정보들을 순회하면서 특정 섬들에 포함된다면 어떤 형태가 될지 예측한다.</li>
 *     <li>만약 순환 구조를 갖는다면, 해당 연결 정보는 버린다.</li>
 *     <li>만약 트리 구조를 갖는다면, 다른 섬들에도 해당 연결 정보로 트리 구조가 되는지 예측해본다.</li>
 *     <li>해당 연결 정보를 병합하고 다른 트리 구조의 섬들도 있다면 병합한다.</li>
 *     <li>예측 결과가 어느 다른 섬들에 속하지 못한다면, 해당 연결 정보는 또 다른 섬들이 된다.</li>
 *     <li>최종적으로 병합된 섬들로부터 연결 비용을 반환한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution(4, new int[][]{{0, 1, 1}, {0, 2, 2}, {1, 2, 5}, {1, 3, 1}, {2, 3, 8}}));
    }

    public static int solution(int n, int[][] costs) {
        int answer = 0;

        List<Link> linkList = new ArrayList<>();
        List<Group> groupList = new ArrayList<>();

        for (int[] c : costs) {
            linkList.add(new Link(c));
        }

        linkList.sort(Comparator.comparingInt(o -> o.cost));

        for (Link link : linkList) {
            if (groupList.isEmpty()) {
                groupList.add(new Group(link));

                continue;
            }

            Group firstGroup = null;
            Group secondGroup = null;
            boolean isCycle = false;

            for (Group group : groupList) {
                Result result = group.predictResult(link);

                if (result == Result.TREE) {
                    if (firstGroup == null) {
                        firstGroup = group;
                    } else {
                        secondGroup = group;

                        break;
                    }
                } else if (result == Result.CYCLE) {
                    isCycle = true;

                    break;
                }
            }

            if (isCycle) {
                continue;
            }

            if (firstGroup == null) {
                groupList.add(new Group(link));
            } else if (secondGroup == null) {
                firstGroup.add(link);
            } else {
                firstGroup.add(link);
                firstGroup.addAll(secondGroup);
                groupList.remove(secondGroup);
            }
        }

        answer = groupList.get(0).costs;

        return answer;
    }

    static class Group {
        Set<Integer> nodeSet = new HashSet<>();
        int costs = 0;

        Group(Link link) {
            add(link);
        }

        Result predictResult(Link link) {
            boolean startNodeFound = nodeSet.contains(link.startNode);
            boolean endNodeFound = nodeSet.contains(link.endNode);

            return startNodeFound && endNodeFound ?
                    Result.CYCLE : (startNodeFound || endNodeFound) ?
                    Result.TREE : Result.OTHER;
        }

        void add(final Link link) {
            nodeSet.add(link.startNode);
            nodeSet.add(link.endNode);

            costs += link.cost;
        }

        void addAll(Group group) {
            nodeSet.addAll(group.nodeSet);

            costs += group.costs;
        }
    }

    enum Result { TREE, CYCLE, OTHER }

    static class Link {
        int startNode;
        int endNode;
        int cost;

        public Link(int[] arr) {
            startNode = arr[0];
            endNode = arr[1];
            cost = arr[2];
        }
    }
}
