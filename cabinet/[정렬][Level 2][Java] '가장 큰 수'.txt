import java.util.*;

/**
 * <h1>가장 큰 수</h1>
 * <h5>분류: 정렬</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>숫자 배열이 주어질 때 가장 큰 수를 만드는 문제이다.</li>
 * <li>숫자들간 중복되는 부분을 제거하고 한 자리씩 대조하여,
 * 앞에 위치할지, 뒤에 위치할지를 결정한다.</li>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution(new int[]{6, 10, 2}));
        System.out.println(solution(new int[]{3, 30, 34, 5, 9}));
    }

    public static String solution(int[] numbers) {
        StringBuilder answer = new StringBuilder();
        Queue<Integer> numberQueue = new PriorityQueue<>((o1, o2) ->
                compare(o1.toString(), o2.toString()));

        for (int n : numbers) {
            numberQueue.add(n);
        }

        if (!numberQueue.isEmpty() && numberQueue.peek() == 0) {
            return "0";
        }

        while (!numberQueue.isEmpty()) {
            answer.append(numberQueue.poll());
        }

        return answer.toString();
    }

    static int compare(String o1, String o2) {
        if (o1.length() == o2.length()) {
            return compareInDetail(o1, o2);
        } else if (o1.startsWith(o2)) {
            return compare(o1.substring(o2.length()), o2);
        } else if (o2.startsWith(o1)) {
            return compare(o1, o2.substring(o1.length()));
        } else {
            return compareInDetail(o1, o2);
        }
    }

    static int compareInDetail(String o1, String o2) {
        for (int i = 0, length = Math.min(o1.length(), o2.length()); i < length; i++) {
            int compared = o2.charAt(i) - o1.charAt(i);

            if (compared != 0) {
                return compared;
            }
        }

        return 0;
    }
}
