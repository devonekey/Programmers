import java.util.*;

/**
 * <h1>전화번호 목록</h1>
 * <h5>분류: 해시</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>주어진 N개의 번호들 중, 하나의 번호가 다른 번호의 접두사로 존재하는지를 찾는 문제이다.</li>
 * <ol>
 *     <li>번호들을 사전적으로 정렬한다.</li>
 *     <li>번호들을 순회하며 직전의 번호 길이만큼 현재의 번호 비교하여 접두사인지 찾는다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public boolean solution(String[] phone_book) {
        boolean answer = true;
        String last = phone_book[0];

        Arrays.sort(phone_book, String::compareTo);

        for (int i = 1; i < phone_book.length; i++) {
            String phone = phone_book[i];

            if (last.length() > phone.length()) {
                last = phone;

                continue;
            } else if (last.equals(phone.substring(0, last.length()))) {
                answer = false;

                break;
            }

            last = phone;
        }

        return answer;
    }

    public static void main(String[] args) {
        System.out.println(new Solution().solution(new String[]{"119", "97674223", "1195524421"}));
        System.out.println(new Solution().solution(new String[]{"123", "456", "789"}));
        System.out.println(new Solution().solution(new String[]{"12", "123", "1235", "567", "88"}));
    }
}
