import java.util.*;

/**
 * <h1>같은 숫자는 싫어</h1>
 * <h5>분류: 스택/큐</h5>
 * <h5>난이도: Level 1</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>배열을 순회하면서 스택에 쌓는데, 이전에 추가한 요소와 중복되는 숫자인 경우,
 * 추가하지 않고 넘어가고 그 스택을 출력하는 문제이다.</li>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        int[] arr;

        arr = solution(new int[]{1, 1, 3, 3, 0, 1, 1});

        for (int e: arr) {
            System.out.print(e + ", ");
        }

        System.out.println();

        arr = solution(new int[]{4, 4, 4, 3, 3});

        for (int e: arr) {
            System.out.print(e + ", ");
        }
    }

    public static int[] solution(int[] arr) {
        int[] answer = {};
        Stack<Integer> stack = new Stack<>();

        for (int num: arr) {
            if (stack.isEmpty() || stack.peek() != num) {
                stack.add(num);
            }
        }

        answer = new int[stack.size()];

        for (int i = 0; i < stack.size(); i++) {
            answer[i] = stack.get(i);
        }

        return answer;
    }
}
