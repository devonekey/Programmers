import java.util.*;

/**
 * <h1>기능 개발</h1>
 * <h5>분류: 스택/큐</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>배열의 데이터가 앞에서 부터 특정 조건을 넘어가면 카운팅하여
 * 모든 데이터들이 조건을 만족할 때 카운팅했던 것들을 출력하는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>'progress'를 증가시킨다.</li>
 *     <li>만약 증가된 'progress'가 조건을 만족하면, 카운팅한다.</li>
 *     <li>다음 'progress'에서 카운팅할 수 없으면, 앞에서 부터 순차적으로 카운팅했던 것들을 저장한다.</li>
 *     <li>카운팅한 것들을 다시 0으로 초기화한다.</li>
 * </ol>
 * <li>출제 의도는 큐를 이용하는 것이지만 어렵게 돌아갔다.</li>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        int[] array;

        array = solution(new int[]{93, 30, 55}, new int[]{1, 30, 5});

        for (int e: array) {
            System.out.print(e + ", ");
        }

        System.out.println();

        array = solution(
                new int[]{95, 90, 99, 99, 80, 99},
                new int[]{1, 1, 1, 1, 1, 1}
        );

        for (int e: array) {
            System.out.print(e + ", ");
        }
    }

    public static int[] solution(int[] progresses, int[] speeds) {
        int[] answer = {};
        int progressesLength = progresses.length;
        int distributedCount = 0;
        int currentCount = 0;
        List<Integer> distributedList = new ArrayList<>();

        do {
            for (int i = distributedCount; i < progressesLength; i++) {
                progresses[i] += speeds[i];

                if (progresses[i] >= 100) {
                    currentCount++;

                    if (distributedCount + currentCount == progressesLength) {
                        distributedList.add(currentCount);

                        distributedCount += currentCount;

                        break;
                    }

                    continue;
                } else if (currentCount > 0 && distributedCount + currentCount == i) {
                    distributedList.add(currentCount);

                    distributedCount += currentCount;
                }

                currentCount = 0;
            }

            currentCount = 0;
        } while(distributedCount < progressesLength);

        int distributedListSize = distributedList.size();
        answer = new int[distributedListSize];

        for (int i = 0; i < distributedListSize; i++) {
            answer[i] = distributedList.get(i);
        }

        return answer;
    }
}
