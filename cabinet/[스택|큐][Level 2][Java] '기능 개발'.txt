import java.util.*;

/**
 * <h1>기능개발</h1>
 * <h5>분류: 스택/큐, 단조 스택</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>배열의 데이터가 앞에서 부터 특정 조건을 넘어가면 카운팅하고,
 * 모든 데이터들이 조건을 만족할 때 카운팅했던 것들을 출력하는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>주어진 데이터를 순회하면서 완료까지 걸리는 날짜를 계산한다.</li>
 *     <li>계산한 날짜가 queue에 쌓인 첫번째 날짜보다 더 크다면, queue의 크기만큼 한꺼번에 배포됐다고 간주한다.</li>
 *     <li>그리고 계산한 날짜를 queue에 추가한다.</li>
 *     <li>위 과정을 반복한다.</li>
 *     <li>모두 순회하고 배포되지 않은 크기만큼 한꺼번에 배포됐다고 간주한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        int[] array;

        array = solution(new int[]{93, 30, 55}, new int[]{1, 30, 5});

        for (int e: array) {
            System.out.print(e + ", ");
        }

        System.out.println();

        array = solution(
                new int[]{95, 90, 99, 99, 80, 99},
                new int[]{1, 1, 1, 1, 1, 1}
        );

        for (int e: array) {
            System.out.print(e + ", ");
        }
    }

    public static int[] solution(int[] progresses, int[] speeds) {
        int[] answer = {};
        Queue<Integer> queue = new LinkedList<>();
        List<Integer> list = new ArrayList<>();
        int remains = progresses.length;

        for (int i = 0; i < progresses.length; i++) {
            int completeDays = (int) Math.ceil((100F - progresses[i]) / speeds[i]);

            if (!queue.isEmpty() && completeDays > queue.peek()) {
                remains -= queue.size();

                list.add(queue.size());
                queue.clear();
            }

            queue.add(completeDays);
        }

        list.add(remains);

        answer = new int[list.size()];

        for (int i = 0; i < list.size(); i++) {
            answer[i] = list.get(i);
        }

        return answer;
    }
}
