import java.util.*;

public class Solution {
    public static void main(String[] args) {
        System.out.println(solution("17"));
        System.out.println(solution("011"));
    }

    static int solution(String numbers) {
        int answer = 0;

        Queue<Character> numberQueue = new PriorityQueue<>(Comparator.reverseOrder());

        for (char c: numbers.toCharArray()) {
            numberQueue.add(c);
        }

        StringBuilder maxNumberBuilder = new StringBuilder();

        while (!numberQueue.isEmpty()) {
            maxNumberBuilder.append(numberQueue.remove());
        }

        int maxNumber = Integer.parseInt(maxNumberBuilder.toString());
        Set<Integer> primeSet = new HashSet<>();

        collectPrime(maxNumber, primeSet);

        Map<Integer, Integer> numberMap = new HashMap<>();

        for (int c : numbers.toCharArray()) {
            int count = numberMap.getOrDefault(c, 0);

            numberMap.put(c, ++count);
        }

        for (int prime : primeSet) {
            Map<Integer, Integer> copiedNumberMap = new HashMap<>(numberMap);
            boolean isContained = true;

            for (int c : Integer.toString(prime).toCharArray()) {
                int count = copiedNumberMap.getOrDefault(c, 0) - 1;

                if (count < 0) {
                    isContained = false;

                    break;
                }

                copiedNumberMap.put(c, count);
            }

            if (isContained) {
                answer++;
            }
        }

        return answer;
    }

    static void collectPrime(int number, Set<Integer> primeSet) {
        primeSet.add(2);

        for (int i = 0; i <= number; i++) {
            if (isPrime(i, primeSet)) {
                primeSet.add(i);
            }
        }
    }

    static boolean isPrime(int number, Set<Integer> primeSet) {
        if (number <= 1) {
            return false;
        } else if (number == 2) {
            return true;
        }

        for (int prime : primeSet) {
            if (number % prime == 0) {
                return false;
            }
        }

        return true;
    }
}
