/**
 * <h1>모음 사전</h1>
 * <h5>분류: 완전탐색</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>주어진 문자열이 사전적으로 몇번째에 위치하는지를 구하는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>현재 문자열에서 특정 위치의 문자를 다음 문자로 교체한다.</li>
 *     <li>주어진 문자열과 교체한 문자열이 같은지 비교한다.</li>
 *     <li>같으면 종료하고, 같지 않으면 다음 위치를 대상으로 위 과정을 반복한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution("AAAAE"));
        System.out.println(solution("AAAE"));
        System.out.println(solution("I"));
        System.out.println(solution("EIO"));
    }

    static String dic = "AEIOU";
    static int count = 0;

    public static int solution(String word) {
        int answer = 0;

        search(word, "", 0);

        answer = count;
        count = 0;

        return answer;
    }

    static boolean search(String word, String current, int index) {
        for (int i = 0; i < dic.length(); i++) {
            current = new StringBuilder(current)
                    .replace(index, index + 1, String.valueOf(dic.charAt(i)))
                    .toString();

            count++;

            if (word.equals(current)) {
                return true;
            }

            boolean found = false;

            if (current.length() < 5) {
                found = search(word, current, index + 1);
            }

            if (found) {
                return true;
            }
        }

        return false;
    }
}
