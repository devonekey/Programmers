import java.util.*;

/**
 * <h1>다리를 지나는 트럭</h1>
 * <h5>분류: 스택/큐</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>트럭이 무게와 길이 제한이 있는 다리를 정해진 순서로 지날 때,
 * 몇초가 소요되는지 묻는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>다리 위를 순회하면서 1초씩 증가시킨다.</li>
 *     <li>다리 위에 트럭이 존재하면 한 칸씩 다리를 지나간다.</li>
 *     <li>다른 트럭이 진입할 수 있다면 진입시킨다.</li>
 *     <li>위 과정을 반복하고 몇초가 소요됐는지 출력한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution(2, 10, new int[]{7,4,5,6}));
        System.out.println(solution(100, 100, new int[]{10}));
        System.out.println(solution(100, 100, new int[]{10, 10, 10, 10, 10, 10, 10, 10, 10, 10}));
    }

    public static int solution(int bridge_length, int weight, int[] truck_weights) {
        int answer = 0;
        Queue<Truck> bridge = new LinkedList<>();
        int lastTruckLocation = 0;
        int totalWeight = 0;
        int truckIndex = 0;

        do {
            answer++;

            if (bridge.isEmpty()) {
                bridge.add(new Truck(truck_weights[truckIndex], bridge_length));

                lastTruckLocation = bridge_length;
                totalWeight += truck_weights[truckIndex];
                truckIndex++;

                continue;
            }

            Truck truck = bridge.peek();

            truck.distance--;
            lastTruckLocation--;

            if (truck.distance == 0) {
                bridge.poll();

                totalWeight -= truck.weight;
            }

            if (truckIndex < truck_weights.length &&
                    totalWeight + truck_weights[truckIndex] <= weight) {
                int distance = bridge_length - lastTruckLocation;

                bridge.add(new Truck(truck_weights[truckIndex], distance));

                lastTruckLocation = bridge_length;
                totalWeight += truck_weights[truckIndex];
                truckIndex++;
            }
        } while (!bridge.isEmpty());

        return answer;
    }

    public static class Truck {
        int weight;
        int distance;

        public Truck(int weight, int distance) {
            this.weight = weight;
            this.distance = distance;
        }
    }
}
