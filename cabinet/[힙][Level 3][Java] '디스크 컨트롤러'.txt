import java.util.*;

/**
 * <h1>디스크 컨트롤러</h1>
 * <h5>분류: 힙</h5>
 * <h5>난이도: Level 3</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>'작업이 요청되는 시점'과 '작업의 소요시간'이 주어질 때,
 * 가장 빨리 처리할 수 있는 작업을 먼저 처리하는 문제이다.</li>
 * <li>데이터를 정렬하는 것과 작업 종료 시간이 연장되면서 추가적으로 포함될 다른 작업들을 찾는 것도 중요하며,
 * 작업을 처리하지 않은 시간이 존재해 이에 관한 것들이 관건이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>'작업이 요청되는 시점' 순서로 큐를 준비하고 초기화 한다.</li>
 *     <li>'작업의 소요시간' 순서로 큐를 준비한다.</li>
 *     <li>큐에서 작업을 하나 꺼내 작업을 처리하기 까지, 그 시간 중에 다른 작업들이 요청되는지 확인하고 있다면 저장한다.</li>
 *     <li>저장했던 작업들을 순차적으로 꺼내면서 작업 시간을 추가하고 작업 종료 시간을 갱신한다.</li>
 *     <li>작업 종료 시간이 갱신됐으므로, 그 시간 중에 다른 작업들이 요청되는지 확인하고 있다면 저장하는 과정을 반복한다.</li>
 *     <li>작업을 처리하지 않은 시간이 존재하여 미처 처리되지 못한 작업들이 있다면, 큐에서 작업을 꺼내고 앞의 과정들을 반복한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        int[][] jobs = {{0, 3}, {1, 9}, {2, 6}};

        System.out.println(solution(jobs));

        jobs = new int[][]{{24, 10}, {28, 39}, {43, 20}, {37, 5}, {47, 22},
                {20, 47}, {15, 34}, {15, 2}, {35, 43}, {26, 1}};

        System.out.println(solution(jobs));
    }

    public static int solution(int[][] jobs) {
        int answer = 0;
        Queue<Job> waitingQueue = new PriorityQueue<>(
                Comparator.comparingInt((Job o) -> o.workingTime)
                        .thenComparingInt(o -> o.index)
        );
        Queue<Job> jobQueue = new PriorityQueue<>(
                Comparator.comparingInt((Job o) -> o.startTime)
                        .thenComparingInt(o -> o.workingTime)
                        .thenComparingInt(o -> o.index)
        );

        for (int i = 0; i < jobs.length; i++) {
            jobQueue.add(new Job(i, jobs[i][0], jobs[i][1]));
        }

        int jobEndTime = 0;

        while (!jobQueue.isEmpty()) {
            if (waitingQueue.isEmpty()) {
                Job firstJob = jobQueue.remove();
                jobEndTime = firstJob.startTime + firstJob.workingTime;
                answer += firstJob.workingTime;
            }

            while (true) {
                for (Job job = jobQueue.peek();
                     job != null && job.startTime < jobEndTime;
                     waitingQueue.add(jobQueue.poll()), job = jobQueue.peek());

                Job job = waitingQueue.poll();

                if (job != null) {
                    answer += jobEndTime - job.startTime + job.workingTime;
                    jobEndTime += job.workingTime;
                } else {
                    break;
                }
            }
        }

        return answer / jobs.length;
    }

    static class Job {
        int index;
        int startTime;
        int workingTime;

        public Job(int index, int startTime, int workingTime) {
            this.index = index;
            this.startTime = startTime;
            this.workingTime = workingTime;
        }
    }
}
