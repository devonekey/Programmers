/**
 * <h1>큰 수 만들기</h1>
 * <h5>분류: 탐욕</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>주어진 숫자열에서 k개 만큼 수를 제거하고 최댓값을 만드는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>시작점은 0으로, 끝지점은 k + 1로 삼는다.</li>
 *     <li>해당 범위 내의 숫자열에서 가장 큰 수를 구한다.</li>
 *     <li>그 수를 기억한다.</li>
 *     <li>구한 수 다음 위치가 시작점이 되고 끝지점은 1을 추가한다.</li>
 *     <li>원하는 길이가 될 때까지 큰 수를 구하는 과정을 반복한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution("1924", 2));
        System.out.println(solution("1231234", 3));
        System.out.println(solution("4177252841", 4));
    }

    public static String solution(String number, int k) {
        String answer = "";
        StringBuilder builder = new StringBuilder();

        for (int beginIndex = 0, endIndex = k + 1;
             endIndex <= number.length();
             endIndex++) {
            String subString = number.substring(beginIndex, endIndex);
            int targetIndex = getHighestNumberIndex(subString);

            builder.append(subString.charAt(targetIndex));

            beginIndex += targetIndex + 1;
        }

        answer = builder.toString();

        return answer;
    }

    private static int getHighestNumberIndex(String head) {
        int max = 0;
        int index = 0;

        for (int i = 0; i < head.length(); i++) {
            int n = head.charAt(i) - '0';

            if (n > max) {
                max = n;
                index = i;
            }
        }

        return index;
    }
}
