 * <li>처음에는 아래와 같은 방법으로 접근했다.

public class Solution {
    public static void main(String[] args) {
        System.out.println("answer: " + solution(new String[][]{
                {"yellow_hat", "headgear"},
                {"blue_sunglasses", "eyewear"},
                {"green_turban", "headgear"},
        }));
        System.out.println("answer: " + solution(new String[][]{
                {"crow_mask", "face"},
                {"blue_sunglasses", "face"},
                {"smoky_makeup", "face"}
        }));
    }

    public static int solution(String[][] clothes) {
        int answer = 1;
//        int answer = 0;
        Map<String, Integer> closet = new HashMap<>();

        for (String[] cloth: clothes) {
            String type = cloth[1];
            int count = closet.getOrDefault(type, 0);

            closet.put(type, ++count);
        }

        List<Integer> values = new ArrayList<>(closet.values());

        for (Integer v : values) {
            answer *= (v + 1);
        }

//        for (int k = 1; k <= values.size(); k++) {
//            answer += combinations(values, values.size(), k, 0, 0);
//        }

        return --answer;
//        return answer;
    }
