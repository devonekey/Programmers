import java.util.*;

/**
 * <h1>타겟 넘버</h1>
 * <h5>분류: 깊이/너비 우선 탐색</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>주어진 배열(numbers)에서 자유롭게 음수 부호를 붙여서 특정 수(target)를 만드는 가짓수를 구하는 문제이다.</li>
 * <li>조합을 이용하여 음수 부호를 붙일 수 있는 모든 경우를 만들면서 그 합이 특정 수와 일치하는지 확인했다.</li>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution(new int[]{1, 1, 1, 1, 1}, 3));
        System.out.println(solution(new int[]{4, 1, 2, 1}, 4));
    }

    public static int solution(int[] numbers, int target) {
        int answer = 0;
        int[] indices;

        for (int i = 1; i <= numbers.length; i++) {
            indices = new int[i];
            answer += combine(numbers, numbers.length, i, indices, 0, 0, target);
        }

        return answer;
    }

    private static int combine(
            int[] array,
            int n,
            int k,
            int[] indices,
            int start,
            int current,
            int target
    ) {
        if (current == k) {
            int[] numbers = Arrays.copyOf(array, n);
            int sum = 0;

            for (int i : indices) {
                numbers[i] = -numbers[i];
            }

            for (int num : numbers) {
                sum += num;
            }

            return sum == target ? 1 : 0;
        }

        int count = 0;

        for (int i = start; i < n; i++) {
            indices[current] = i;
            count += combine(array, n, k, Arrays.copyOf(indices, k), i + 1, current + 1, target);
        }

        return count;
    }
}
