import java.util.*;

/**
 * <h1>전력망을 둘로 나누기</h1>
 * <h5>분류: 완전탐색</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>트리 형태로 이어진 전력망을 둘로 나누었을 때,
 * 두 전력망의 송전탑 차이 최솟값을 구하는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>하나의 전선을 선택하여 두 전력망으로 나눈다.</li>
 *     <li>모든 전선을 큐로 초기화한다.</li>
 *     <li>큐를 순회하면서 전력망에 인접한 송전탑을 찾아 포함시킨다.</li>
 *     <li>아직 어느 그룹에도 속하지 못한 전선은 다시 큐에 집어 넣는다.</li>
 *     <li>큐가 비었으면 두 전력망의 송전탑 차이를 구하고 최솟값을 취한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution(9, new int[][]{{1, 3}, {2, 3}, {3, 4}, {4, 5}, {4, 6}, {4, 7}, {7, 8}, {7, 9}}));
        System.out.println(solution(4, new int[][]{{1, 2}, {2, 3}, {3, 4}}));
        System.out.println(solution(7, new int[][]{{1, 2}, {2, 7}, {3, 7}, {3, 4}, {4, 5}, {6, 7}}));
    }

    public static int solution(int n, int[][] wires) {
        int answer = -1;
        List<Wire> wireList = new ArrayList<>();
        answer = wires.length;

        for (int[] wire : wires) {
            wireList.add(new Wire(wire[0], wire[1]));
        }

        for (int i = 0; i < wires.length; i++) {
            Set<Integer> firstGroup = new HashSet<>();
            Set<Integer> secondGroup = new HashSet<>();
            int first = wireList.get(i).v1;
            int second = wireList.get(i).v2;

            firstGroup.add(first);
            secondGroup.add(second);

            Queue<Wire> queue = new LinkedList<>(wireList);

            do {
                Wire wire = queue.remove();

                if (wire.v1 == first && wire.v2 == second) {
                    continue;
                }

                if (firstGroup.contains(wire.v1)) {
                    firstGroup.add(wire.v2);
                } else if (firstGroup.contains(wire.v2)) {
                    firstGroup.add(wire.v1);
                } else if (secondGroup.contains(wire.v1)) {
                    secondGroup.add(wire.v2);
                } else if (secondGroup.contains(wire.v2)) {
                    secondGroup.add(wire.v1);
                } else {
                    queue.add(wire);
                }
            } while (!queue.isEmpty());

            answer = Math.min(answer, Math.abs(firstGroup.size() - secondGroup.size()));
        }

        return answer;
    }

    static class Wire {
        int v1;
        int v2;

        public Wire(int v1, int v2) {
            this.v1 = v1;
            this.v2 = v2;
        }
    }
}
