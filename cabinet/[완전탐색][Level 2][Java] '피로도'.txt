/**
 * <h1>피로도</h1>
 * <h5>분류: 완전탐색</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>현재 피로도 내에서 방문할 수 있는 던전들의 최대 수를 찾는 문제이다.</li>
 * <li>'최소 필요 피로도'와 '소모 피로도' 중 어느 하나의 조건을 기준으로는 최적의 답을 도출할 수 없어서,
 * 순회를 이용하여 모든 조건을 탐색하는 방식으로 답을 도출했다.</li>
 * <li>동적 계획법 등으로도 해결할 수 있는 등 다양한 풀이가 있다.</li>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution(80, new int[][]{{80, 20}, {50, 40}, {30, 10}}));
    }

    static int answer = 0;

    public static int solution(int k, int[][] dungeons) {
        permute(dungeons, 0, dungeons.length - 1, k);

        return answer;
    }

    static void permute(int[][] array, int l, int r, int k) {
        if (l == r) {
            int count = 0;

            for (int[] e: array) {
                if (e[0] <= k && e[1] <= k) {
                    k -= e[1];

                    count++;
                }
            }

            answer = Math.max(count, answer);

            return;
        }

        for (int i = l; i <= r; i++) {
            int[] temp = array[i];
            array[i] = array[l];
            array[l] = temp;

            permute(array, l + 1, r, k);

            temp = array[i];
            array[i] = array[l];
            array[l] = temp;
        }
    }
}
