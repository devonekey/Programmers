import java.util.*;

/**
 * <h1>H-Index</h1>
 * <h5>분류: 정렬</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>인용이 가장 많이 된 논문부터 0까지 순회하면서 특정 횟수 이상의 논문 수가
 * 나머지 논문들보다 더 많거나 같을 때 그 특정 횟수를 출력하는 문제이다.</li>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution(new int[]{3, 0, 6, 1, 5}));
        System.out.println(solution(new int[]{0}));
        System.out.println(solution(new int[]{3, 4}));
        System.out.println(solution(new int[]{1, 2, 3, 5, 6, 7, 10, 11}));
        System.out.println(solution(new int[]{3, 5, 11, 6, 1, 5, 3, 3, 1, 41}));
        System.out.println(solution(new int[]{1, 11, 111, 1111}));
    }

    static int solution(int[] citations) {
        int answer = 0;
        Queue<Integer> citationQueue = new PriorityQueue<>((o1, o2) -> Integer.compare(o2, o1));

        for (int c: citations) {
            citationQueue.add(c);
        }

        Integer c;

        for (int h = citationQueue.element(); h >= 0; h--) {
            c = citationQueue.peek();

            do {
                if (c != null && c >= h) {
                    c = citationQueue.poll();
                } else {
                    break;
                }
            } while (!citationQueue.isEmpty() && citationQueue.peek() >= h);

            if (citations.length - citationQueue.size() >= h
                    && citationQueue.size() <= h) {
                answer = h;

                return answer;
            }
        }

        return answer;
    }
}
