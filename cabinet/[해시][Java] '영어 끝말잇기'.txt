import java.util.*;

/**
 * <h1>영어 끝말잇기</h1>
 * <h5>분류: 해시</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>사람 수와 각 단어가 들어있는 배열이 주어지고
 * 끝말잇기를 하다 중복되는 단어를 말하거나 끝말잇기가 성립되지 않을 때,
 * 탈락한 사람의 번호와 탈락한 사람이 몇 번째에 탈락하는지를 구하는 문제이다.</li>
 * <li>배열을 모두 순회해도 탈락한 사람이 없을 수도 있다.</li>
 * <ol>
 *     <li>주어진 배열을 순회한다.</li>
 *     <li>다음 사람의 단어가 등장하지 않았으면서 단어 첫 번째 알파벳이 직전 단어의 마지막 알파벳과 일치하면, 통과한다.</li>
 *     <li>위의 조건에 부합하지 않은 단어는 탈락으로 간주하고 번호와 몇 번째 탈락인지 구한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public int[] solution(int n, String[] words) {
        int[] answer = new int[2];
        String last = words[0];
        Set<String> set = new HashSet<>();

        set.add(words[0]);

        for (int i = 1; i < words.length; i++) {
            String word = words[i];

            if (!set.contains(word) && word.charAt(0) == last.charAt(last.length() - 1)) {
                last = word;

                set.add(word);

                continue;
            }

            answer = new int[]{(i % n) + 1, (i / n) + 1};

            break;
        }

        return answer;
    }

    public static void main(String[] args) {
        int[] result;

        result = new Solution().solution(3, new String[]{
                "tank", "kick", "know",
                "wheel", "land", "dream",
                "mother", "robot", "tank"
        });

        System.out.println("[" + result[0] + ", " + result[1] + "]");

        result = new Solution().solution(5, new String[]{
                "hello", "observe", "effect", "take", "either",
                "recognize", "encourage", "ensure", "establish", "hang",
                "gather", "refer", "reference", "estimate", "executive"
        });

        System.out.println("[" + result[0] + ", " + result[1] + "]");

        result = new Solution().solution(2, new String[]{
                "hello", "one",
                "even", "never",
                "now", "world",
                "draw"
        });

        System.out.println("[" + result[0] + ", " + result[1] + "]");
    }
}
