import java.util.*;

/**
 * <h1>프로세스</h1>
 * <h5>분류: 스택/큐</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>우선순위가 높은 프로세스부터 실행했을 때,
 * 특정 위치의 프로세스는 몇 번째로 실행됐는지를 묻는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>주어진 우선순위를 내림차순으로 정렬한 결과를 따로 저장한다.</li>
 *     <li>정렬된 우선순위를 하나씩 꺼내면서 주어진 우선순위와 비교하여 정렬된 우선순위를 발견하면,
 *     정렬된 우선순위에서 제외시킨다.</li>
 *     <li>주어진 우선순위가 정렬된 우선순위보다 작은 경우, 맨 뒤로 보낸다.</li>
 *     <li>제외시키고 보내는 과정을 반복하고, 특정 위치의 프로세스가 몇 번째로 실행됐는지 출력한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        System.out.println(solution(new int[]{2, 1, 3, 2}, 2));
        System.out.println(solution(new int[]{1, 1, 9, 1, 1, 1}, 0));
    }

    public static int solution(int[] priorities, int location) {
        int answer = 0;
        Queue<Process> processQueue = new LinkedList<>();
        List<Integer> sortedPriorities = new ArrayList<>();

        for (int i = 0; i < priorities.length; i++) {
            processQueue.add(new Process(i, priorities[i]));
            sortedPriorities.add(priorities[i]);
        }

        sortedPriorities.sort((o1, o2) -> Integer.compare(o2, o1));

        List<Process> finishedProcessList = new ArrayList<>();

        do {
            int highestPriority = sortedPriorities.get(0);

            for (Process process: processQueue) {
                if (process.priority == highestPriority) {
                    finishedProcessList.add(processQueue.poll());
                    sortedPriorities.remove(0);

                    break;
                } else if (process.priority < highestPriority) {
                    processQueue.add(processQueue.poll());

                    break;
                }
            }
        } while (!sortedPriorities.isEmpty());

        for (int i = 0; i < finishedProcessList.size(); i++) {
            if (location == finishedProcessList.get(i).location) {
                answer = i + 1;

                break;
            }
        }

        return answer;
    }

    static class Process {
        int location;
        int priority;

        public Process(int location, int priority) {
            this.location = location;
            this.priority = priority;
        }
    }
}
