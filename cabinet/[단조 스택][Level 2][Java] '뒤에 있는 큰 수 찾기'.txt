import java.util.*;

/**
 * <h1>뒤에 있는 큰 수 찾기</h1>
 * <h5>분류: 단조 스택</h5>
 * <h5>난이도: Level 2</h5>
 * <h5>풀이</h5>
 * <ul>
 * <li>배열이 주어질 때, 뒷 큰수를 찾아 그 값을 새 배열에 기록하고 반환하는 문제이다.</li>
 * <li>아래와 같은 방법으로 해결했다.</li>
 * <ol>
 *     <li>배열을 정방향으로 순회한다.</li>
 *     <li>stack이 비어있지 않다면, 현재 값과 stack 최상위의 값이 가기키는 값을 비교한다.</li>
 *     <li>현재 값이 크다면 stack에서 값을 꺼내고, 그 값이 가리키는 새 배열의 위치에 현재 값을 넣는다.</li>
 *     <li>위 과정을 stack이 모두 비워지거나 현재 값이 작을 때까지 반복한다.</li>
 *     <li>stack에 현재 값의 위치를 추가한다.</li>
 *     <li>위 과정을 주어진 배열의 길이 만큼 반복한다.</li>
 * </ol>
 * </ul>
 */
public class Solution {
    public static void main(String[] args) {
        int[] answer = solution(new int[]{2, 3, 4, 5});

        for (int a : answer) {
            System.out.print(a + ", ");
        }

        System.out.println();

        answer = solution(new int[]{9, 1, 5, 3, 6, 2});

        for (int a : answer) {
            System.out.print(a + ", ");
        }
    }

    public static int[] solution(int[] numbers) {
        int[] answer = new int[numbers.length];
        Stack<Integer> indices = new Stack<>();

        for (int i = 0; i < numbers.length; i++) {
            answer[i] = -1;
        }

        for (int i = 0; i < numbers.length; i++) {
            int number = numbers[i];

            while (!indices.isEmpty() && number > numbers[indices.peek()]) {
                answer[indices.pop()] = number;
            }

            indices.push(i);
        }

        return answer;
    }
}
